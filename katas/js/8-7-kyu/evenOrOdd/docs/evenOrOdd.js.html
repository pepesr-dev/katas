//Toma un entero como argumento y retorna Even or Odd

/**
 * Determina si un número entero es par(even) o impar(odd)
 * @param {number} num - El número entero a evaluar
 * @returns {string} Retorna "Even" si el número es par o "Odd" si el número es impar
 */
function evenOrOdd(num) {
  return num % 2 === 0 ? "Even" : "Odd";
}

//Exportar la función
module.exports = evenOrOdd;
